cmake_minimum_required(VERSION 3.16...3.27)
project(NORMPhosphogypsum CXX)
find_package(Geant4 11.0 REQUIRED)
include(${Geant4_USE_FILE})
file(GLOB sources ${PROJECT_SOURCE_DIR}/src/*.cc)
file(GLOB headers ${PROJECT_SOURCE_DIR}/include/*.hh)
add_executable(norm norm.cc ${sources} ${headers})
target_include_directories(norm PRIVATE ${PROJECT_SOURCE_DIR}/include)
target_link_libraries(norm ${Geant4_LIBRARIES})
file(COPY ${PROJECT_SOURCE_DIR}/macros DESTINATION ${PROJECT_BINARY_DIR})
